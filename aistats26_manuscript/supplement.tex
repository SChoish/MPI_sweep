\section{PROOFS AND ADDITIONAL ANALYSIS}
\label{app:proofs}

\subsection{Behavior-anchor decomposition and Proposition~\ref{prop:prox_equiv}}
For any square-integrable random action $a$ conditioned on $s$, let
$b(s)=\mathbb E[a\mid s]$. The bias--variance identity gives
\begin{equation}
\begin{split}
\mathbb E[\|\mu(s)-a\|_2^2\mid s]
={}&\|\mu(s)-b(s)\|_2^2\\
&+\mathbb E[\|a-b(s)\|_2^2\mid s].
\end{split}
\end{equation}
Averaging over states and dividing by $d$ proves
Eq.~\eqref{eq:bc_decomposition}; the second term is independent of $\mu$.

For state-conditioned deterministic policies
$\pi_\mu(\cdot\mid s)=\delta_{\mu(s)}$ and
$\pi_\nu(\cdot\mid s)=\delta_{\nu(s)}$,
\begin{equation}
W_2^2(\pi_\mu(\cdot\mid s),\pi_\nu(\cdot\mid s))
=\|\mu(s)-\nu(s)\|_2^2.
\end{equation}
Consequently, $\mathsf d_d^2$ is a fixed rescaling of the integrated squared
Wasserstein distance. The normalized anchored objective is
\begin{equation}
-\alpha\mathbb E_s\bar Q(s,\mu(s))
+\mathsf d_d^2(\mu,\nu).
\end{equation}
Multiplication by the positive constant $1/\alpha$ identifies it with
Eq.~\eqref{eq:prox_operator} when $1/(2h)=1/\alpha$, or $h=\alpha/2$. If the value term
is instead $-\lambda\mathbb E_s\widehat Q$, then
$\lambda\widehat Q=\lambda C\bar Q$ and $h=\lambda C/2$.

\subsection{Proof of Proposition~\ref{prop:margin}}
By $\varepsilon_k$-optimality relative to the feasible comparator
$\mu_{k-1}$,
\begin{equation}
\begin{split}
-&\frac{2h_k}{C_k}\mathbb E_s\widehat Q(s,\mu_k(s))
+\mathsf d_d^2(\mu_k,\mu_{k-1})\\
&\leq
-\frac{2h_k}{C_k}\mathbb E_s\widehat Q(s,\mu_{k-1}(s))
+\varepsilon_k.
\end{split}
\end{equation}
Rearranging yields Eq.~\eqref{eq:margin}.

\subsection{Critic-error transfer}
Let $\widehat Q=Q^\dagger+e$. Substitution into Eq.~\eqref{eq:margin} gives
\begin{equation}
\begin{split}
m_k^{\widehat Q}
={}&\frac{2h_k}{C_k}\mathbb E_s
 [Q^\dagger(\mu_k)-Q^\dagger(\mu_{k-1})]
 -\mathsf d_d^2(\mu_k,\mu_{k-1})\\
&+\frac{2h_k}{C_k}\mathbb E_s[e(\mu_k)-e(\mu_{k-1})]\\
={}&m_k^{Q^\dagger}
+\frac{2h_k}{C_k}\mathbb E_s[e(\mu_k)-e(\mu_{k-1})].
\end{split}
\end{equation}
This proves Eq.~\eqref{eq:critic_error_margin}. If
$|e(s,a)|\leq\delta_k$ on the realized path, then
\begin{equation}
m_k^{Q^\dagger}\geq-\varepsilon_k-\frac{4h_k}{C_k}\delta_k.
\end{equation}
The result concerns a trusted critic objective. A true-return statement would
require additional occupancy and value-estimation assumptions.

\subsection{Proof of Proposition~\ref{prop:exposure}}
Let $F_{s'}(a)=\min_jQ_j^-(s',a)$. If every
$Q_j^-(s',\cdot)$ is $L(s')$-Lipschitz, then
\begin{equation}
\begin{split}
|F_{s'}(a)-F_{s'}(b)|
&\leq\max_j|Q_j^-(s',a)-Q_j^-(s',b)|\\
&\leq L(s')\|a-b\|_2.
\end{split}
\end{equation}
Therefore,
\begin{equation}
|y_{\mu_1}-y_{\mu_0}|
\leq\gamma L(s')\|\mu_1(s')-\mu_0(s')\|_2.
\end{equation}
Squaring, averaging over dataset transitions, and taking the square root proves
Eq.~\eqref{eq:exposure_bound}. Later actors do not occur because the TD target
conditions only on $\mu_1$.

\subsection{Local explicit--implicit equivalence}
Let $\beta=dh/C$, $g=\nabla_a\widehat Q(a_0)$, and suppose the action
constraint is inactive. The action-metric-matched linearized update is
$\Delta a_{\rm lin}=\beta g$. A statewise exact proximal solution satisfies
\begin{equation}
\Delta a_{\rm prox}
=\beta\nabla_a\widehat Q(a_0+\Delta a_{\rm prox}).
\end{equation}
If the Hessian of $\widehat Q$ is locally Lipschitz near $a_0$,
\begin{equation}
\nabla_a\widehat Q(a_0+\Delta a)
=g+H\Delta a+O(\|\Delta a\|^2).
\end{equation}
Thus
\begin{equation}
(I-\beta H)\Delta a_{\rm prox}
=\beta g+O(\beta\|\Delta a_{\rm prox}\|^2).
\end{equation}
For $\beta\|H\|<1$ and sufficiently small $\beta$,
\begin{equation}
\Delta a_{\rm prox}=\beta g+\beta^2Hg+O(\beta^3),
\end{equation}
whereas $\Delta a_{\rm lin}=\beta g$. With only $C^2$ regularity, the safe
remainder is $o(\beta^2)$ rather than $O(\beta^3)$. Both ideal updates are
first-order schemes for fixed positive $C$, and changing a fixed $C$ only
rescales time. A positive
policy-dependent global $C$ preserves the raw-critic ascent direction under an
adaptive time parameterization in the infinitesimal limit. This derivation is
not applied to the large-budget regime with active projection, ReLU-region
crossings, critic drift, or inaccurate actor solves.

\subsection{Proof of Proposition~\ref{prop:local_composition}}
Write $x_j=x_0+jhf_0+O(h^2)$ for fixed $K$, which follows inductively from the
one-step consistency assumption. Smoothness then gives
\begin{equation}
f(x_j)=f_0+jhJ_0f_0+O(h^2).
\end{equation}
Summing the $K$ increments,
\begin{equation}
\begin{aligned}
x_K-x_0
={}&h\sum_{j=0}^{K-1}f(x_j)
+h^2\sum_{j=0}^{K-1}p(x_j)+O(Kh^3)\\
={}&Khf_0+h^2\frac{K(K-1)}{2}J_0f_0\\
&+Kh^2p_0+O(h^3).
\end{aligned}
\end{equation}
where the final remainder uses fixed $K$. Substituting $h=T/K$ proves
Eq.~\eqref{eq:k_comp}. The exact flow satisfies
$x(T)=x_0+Tf_0+\tfrac12T^2J_0f_0+O(T^3)$. Thus explicit Euler has leading
error $-T^2J_0f_0/(2K)$, while backward Euler has
$+T^2J_0f_0/(2K)$. Both remain first-order methods.

\section{EXPERIMENTAL DETAILS}
\label{app:experiments}

\subsection{Algorithms and schedules}
The implementation parameter is $T=\tau=\alpha/2$, and the normalized
budget is split as $h=T/K$. Thus $K=1$ recovers TD3+BC with coefficient
$\alpha=2T$. BAR-Prox uses Eq.~\eqref{eq:actorloss} with a moving reference;
BAR-Lin constructs Eq.~\eqref{eq:linearized_target} and uses MSE regression.
The $K$ actor networks are independently initialized persistent networks with
separate Adam states. Each iteration first updates the online critic. On every second iteration,
one optimizer step is then performed per actor in chain order on the same
minibatch: actor one updates first; its target actor and the updated critic each
receive a Polyak update of $.005$; later actors then use stop-gradient
references from the updated online chain. Equations~\eqref{eq:qnorm} and the main-text linearized scale define the
actual normalization, with safeguard $10^{-6}$. All methods share data, critic
architecture, and $10^6$ critic updates, while a $K$-hop run performs $K$ actor
optimizer steps and is not compute matched.

The ideal proximal map is not a certificate for this live chain. For a later
persistent actor that differs from its predecessor, even $h=0$ leaves
\begin{equation}
\nabla_{\theta_k}\mathbb E_s\!
\left[\|\mu_k(s)-\sg\mu_{k-1}(s)\|_2^2/d\right]\ne0
\end{equation}
in general. One Adam step therefore need not become a near-identity map as the
nominal coefficient vanishes; the ideal analysis supplies local geometry, while
the sweep evaluates the implemented optimizer trajectory.

We use a 256--256 ReLU actor MLP with tanh output and twin 256--256 ReLU critic MLPs, batch size 256, Adam with
learning rate $3\times10^{-4}$, discount $.99$, policy delay 2, target noise
$.2$ and clip $.5$ times the action bound, and state normalization by dataset
mean and standard deviation plus $10^{-3}$. Every reported sweep score is the
ten-episode D4RL normalized mean at the final $10^6$-update checkpoint.
Archived configurations record 50,000-update intermediate evaluation for
K1/P2/P3, mixed 50,000/final-only evaluation for L2, and final-only evaluation
for L3; exact K4 run configurations were not retained. Intermediate cadence
does not enter the headline estimand. The 14-value grid spans $.05$ to $20$.
Proximal procedures use seeds 0--3, while linearized procedures use seeds 0--1.

The nine exact datasets are the Cartesian product of domains
\texttt{halfcheetah}, \texttt{hopper}, and \texttt{walker2d} with suffixes
\texttt{medium-v2}, \texttt{medium-replay-v2}, and \texttt{expert-v2}.
Timeout rows are omitted from transition construction; terminal flags stop
bootstrapping.

\subsection{Diagnostic quantities}
On a fixed current-state validation batch with paired dataset actions, we
compute the empirical action-averaged proxies
\[
\begin{aligned}
\widehat D_{\rm critic}
&=\mathbb E_{(s,a)\sim\mathcal D}\frac{1}{d}\|\mu_1(s)-a\|_2^2,\\
\widehat D_{\rm final}
&=\mathbb E_{(s,a)\sim\mathcal D}\frac{1}{d}\|\mu_K(s)-a\|_2^2.
\end{aligned}
\]
These quantities measure the online actor chain relative to its sample anchor.
They are proxies for, rather than direct measurements of, the next-state
target-policy displacement in Proposition~\ref{prop:exposure}: the practical
TD target uses a Polyak target actor and policy-smoothing noise.
By Eq.~\eqref{eq:bc_decomposition}, the population expectation of each proxy
equals squared displacement from the conditional-mean behavior map plus the
same additive conditional-action variance. Population differences therefore
cancel this common floor, but ratios do not: a ratio of sample-anchored proxies
is not a ratio of mean-policy displacements. Finite validation batches add
sampling error to both interpretations.
We additionally evaluate the deterministic Polyak target actor and 64 draws
of clipped target-smoothing noise on 4096 next-state transitions per
checkpoint. For source index $i$, the audit pairs $s'_i=s_{i+1}$ with dataset
action $a_{i+1}$; it excludes timeout rows and excludes true terminal rows from
this bootstrap-only sample. All target-exposure quantities remain squared
displacement per action coordinate relative to the paired next dataset action. Thus absolute
levels include conditional action variance, whereas paired differences use a
common sampled anchor.
For BAR-Lin, projection attenuation is the RMS norm ratio between the clipped
and raw target displacements. If $\widetilde g$ is the normalized critic action
gradient and $\Delta a$ is the learned actor displacement, define
\[
h_{\rm eff}:=
\frac{\mathbb E\langle\Delta a,\widetilde g\rangle}
{\mathbb E\|\widetilde g\|_2^2}.
\]
We report $h_{\rm eff}/(dT/K)$ as the effective fraction of the nominal
linear-target coefficient. The matched audit evaluates $512$ fixed states per environment--seed run. We also record action saturation, TD-error quantiles, critic magnitude, critic loss, and gains under both the jointly trained critic and a controlled common critic: the frozen final TD3+BC $T=1$ critic from a separate canonical reference run outside the fourteen-value sweep, matched by environment and seed. A run-total gain is the sum of mean $Q$ changes across the $K$ hops on this shared batch. The common critic supports cross-critic comparison but is not treated as ground truth. A run is descriptively labeled collapsed only from its external normalized return, using the threshold $20$.

For the exploratory simulator-reference audit, we clone MuJoCo integration
state and environment RNG from evaluation-policy trajectories, record the
episode clock, and reset the TimeLimit for a continuing-task critic-calibration
estimand. At each state we compare $a_1=\mu_1(s)$ and
$a_K=\mu_K(s)$ under continuations $\rho_1$, $\mu_1$, and $\mu_K$, where
$\rho_1$ is the Polyak first target actor with TD3 clipped target-smoothing
noise. The primary calibration target is the discounted return under this
effective backup continuation. The screening covers three environments,
$T\in\{4,7,10,14,20\}$, $K\in\{2,3\}$, two seeds, and 12 states per
checkpoint. It uses a 1000-step horizon, $\gamma=.99$, four common-random-number
target-smoothing rollouts, and a one-return-unit ranking dead zone. It is a
post-hoc screening study, not a causal routing intervention.

The route-only intervention trains two actor-detached shadow critics from identical initial parameters on the same minibatches and target-smoothing draws while the primary first-hop critic continues to drive the shared actor trajectory. The four environment--budget cells---Hopper-medium at $T\in\{10,20\}$ and Walker2d-medium at $T\in\{4,7\}$, each with seeds 0--1---were selected post hoc from visible stability-frontier transitions, so inference is descriptive of this targeted sample. One shadow uses the first-hop target actor and the other the final-hop target actor. The simulator audit evaluates 12 cloned states at four checkpoints in each of eight runs; inference aggregates checkpoint-level state medians to the training-run level.

The compute-matched study uses the same eight final cells. Sequential $K=3$ performs actor updates anchored at $\mu_0,\mu_1,$ and $\mu_2$; the fixed-reference control anchors both later actors at $\mu_1$; the direct-final control applies two optimizer calls to one $2T/3$ objective anchored at $\mu_1$. Every condition performs three actor optimizer calls per delayed update. The target-lag control instead keeps $K=1$ and changes only the actor-target Polyak rate from $.005$ to $.005/3$, leaving the critic-target rate unchanged.

The frozen-critic small-step audit uses the final TD3+BC $T=1$ critic for all nine environments and two seeds. At 512 interior dataset actions per run it compares the projected explicit step with a projected fixed-point solution of the implicit action equation for $h\in\{.001,.003,.01,.03,.1\}$. This is an action-level local-consistency test, not a return or coupled-training experiment.

\subsection{Estimands and uncertainty}
The primary high-budget estimand uses the upper seven values of the ordered
fourteen-point grid, $T\ge4$. Within each task it averages the available seeds
and those seven budgets, then weights the nine task means equally. This is an
operational stress-region aggregate, not a claim of pointwise monotonicity in
$T$ or task. Moving the boundary to $T\ge10$ preserves the strict proximal
ordering: $20.27,31.01,42.23,56.79$ for $K=1,2,3,4$. A seed-mean environment--budget cell is called collapsed when its
normalized score is below 20. Thresholds 0, 10, 20, 30, and 40 expose
sensitivity to that descriptive boundary.

For score contrasts we resample the nine task-level high-budget means with
replacement for 100,000 draws using analysis seed 20260830. Raw-run collapse
risk is first averaged within each task and then resampled by task using the
same draw count and seed. These percentile intervals preserve the complete
within-task budget and seed structure but do not remove dependence among the
three datasets sharing one dynamics family. Cross-$K$ means estimate each
randomized procedure's marginal performance. Identical critic initialization
across depth is not required for validity; common random numbers would reduce
variance. Paired rescue counts remain descriptive under the realized seed
coupling.

The paired route-only study uses the same eight runs and aggregates four
checkpoints per run. For route $r\in\{\mathrm{first},\mathrm{final}\}$, its
statewise symmetric relative error against the common first-route continuation
is
\begin{equation}
e_r(s)=\frac{|Q_r(s,a_K)-Q_{\rm MC}^{\rho_1}(s,a_K)|}
{|Q_r(s,a_K)|+|Q_{\rm MC}^{\rho_1}(s,a_K)|+10^{-12}},
\qquad \Delta(s)=e_{\rm final}(s)-e_{\rm first}(s).
\end{equation}
The primary statistic takes the state median of $\Delta$ within each checkpoint
and then averages the four checkpoints within a run. All eight run-level
checkpoint-aggregate differences are positive. The direction holds in only five of eight comparisons
when restricted to the final checkpoint. Under the primary aggregation, the mean
is $.124$, median is $.013$, exact two-sided sign-flip $p=.0078$, and
run-bootstrap mean 95\% interval is $[.008,.323]$. Because the routes imply
different continuation policies, this estimates relative alignment with the
primary first-route value estimand rather than own-continuation calibration; it
is not a return-effect estimate.

In the frozen-critic audit, 15 of 18 environment--seed runs have at least one converged, projection-free point eligible for the local comparison. Every eligible run has maximum local relative explicit--implicit discrepancy below $2.7\%$; the median run maximum is $.55\%$. Three runs have no eligible point, and a log--log slope is identifiable in only one run because a frozen ReLU critic can produce floating-point agreement inside one activation region. The audit supports local numerical agreement but does not estimate a universal $O(h^2)$ rate.
