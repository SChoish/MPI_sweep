\section{Proofs and Additional Analysis}
\label{app:proofs}

\subsection{Behavior-anchor decomposition and Proposition~\ref{prop:spi_equiv}}
For any square-integrable random action $a$ conditioned on $s$, let
$b(s)=\mathbb E[a\mid s]$. The bias--variance identity gives
\begin{equation}
\begin{split}
\mathbb E[\|\mu(s)-a\|_2^2\mid s]
={}&\|\mu(s)-b(s)\|_2^2\\
&+\mathbb E[\|a-b(s)\|_2^2\mid s].
\end{split}
\end{equation}
Averaging over states and dividing by $d$ proves
Eq.~\eqref{eq:bc_decomposition}; the second term is independent of $\mu$.

For state-conditioned deterministic policies
$\pi_\mu(\cdot\mid s)=\delta_{\mu(s)}$ and
$\pi_\nu(\cdot\mid s)=\delta_{\nu(s)}$,
\begin{equation}
W_2^2(\pi_\mu(\cdot\mid s),\pi_\nu(\cdot\mid s))
=\|\mu(s)-\nu(s)\|_2^2.
\end{equation}
Consequently, $\mathsf d_d^2$ is a fixed rescaling of the integrated squared
Wasserstein distance. The normalized anchored objective is
\begin{equation}
-\alpha\mathbb E_s\bar Q(s,\mu(s))
+\mathsf d_d^2(\mu,\nu).
\end{equation}
Multiplication by the positive constant $1/\alpha$ identifies it with
Eq.~\eqref{eq:spi} when $1/(2h)=1/\alpha$, or $h=\alpha/2$. If the value term
is instead $-\lambda\mathbb E_s\widehat Q$, then
$\lambda\widehat Q=\lambda C\bar Q$ and $h=\lambda C/2$.

\subsection{Proof of Proposition~\ref{prop:margin}}
By $\varepsilon_k$-optimality relative to the feasible comparator
$\mu_{k-1}$,
\begin{equation}
\begin{split}
-&\frac{2h_k}{C_k}\mathbb E_s\widehat Q(s,\mu_k(s))
+\mathsf d_d^2(\mu_k,\mu_{k-1})\\
&\leq
-\frac{2h_k}{C_k}\mathbb E_s\widehat Q(s,\mu_{k-1}(s))
+\varepsilon_k.
\end{split}
\end{equation}
Rearranging yields Eq.~\eqref{eq:margin}.

\subsection{Critic-error transfer}
Let $\widehat Q=Q^\dagger+e$. Substitution into Eq.~\eqref{eq:margin} gives
\begin{equation}
\begin{split}
m_k^{\widehat Q}
={}&\frac{2h_k}{C_k}\mathbb E_s
 [Q^\dagger(\mu_k)-Q^\dagger(\mu_{k-1})]
 -\mathsf d_d^2(\mu_k,\mu_{k-1})\\
&+\frac{2h_k}{C_k}\mathbb E_s[e(\mu_k)-e(\mu_{k-1})]\\
={}&m_k^{Q^\dagger}
+\frac{2h_k}{C_k}\mathbb E_s[e(\mu_k)-e(\mu_{k-1})].
\end{split}
\end{equation}
This proves Eq.~\eqref{eq:critic_error_margin}. If
$|e(s,a)|\leq\delta_k$ on the realized path, then
\begin{equation}
m_k^{Q^\dagger}\geq-\varepsilon_k-\frac{4h_k}{C_k}\delta_k.
\end{equation}
The result concerns a trusted critic objective. A true-return statement would
require additional occupancy and value-estimation assumptions.

\subsection{Proof of Proposition~\ref{prop:exposure}}
Let $F_{s'}(a)=\min_jQ_j^-(s',a)$. If every
$Q_j^-(s',\cdot)$ is $L(s')$-Lipschitz, then
\begin{equation}
\begin{split}
|F_{s'}(a)-F_{s'}(b)|
&\leq\max_j|Q_j^-(s',a)-Q_j^-(s',b)|\\
&\leq L(s')\|a-b\|_2.
\end{split}
\end{equation}
Therefore,
\begin{equation}
|y_{\mu_1}-y_{\mu_0}|
\leq\gamma L(s')\|\mu_1(s')-\mu_0(s')\|_2.
\end{equation}
Squaring, averaging over dataset transitions, and taking the square root proves
Eq.~\eqref{eq:exposure_bound}. Later actors do not occur because the TD target
conditions only on $\mu_1$.

\subsection{Local explicit--implicit equivalence}
Let $\beta=dh/C$, $g=\nabla_a\widehat Q(a_0)$, and suppose the action
constraint is inactive. The scale-matched linearized update is
$\Delta a_{\rm lin}=\beta g$. A statewise exact proximal solution satisfies
\begin{equation}
\Delta a_{\rm prox}
=\beta\nabla_a\widehat Q(a_0+\Delta a_{\rm prox}).
\end{equation}
If $\widehat Q$ is twice continuously differentiable near $a_0$,
\begin{equation}
\nabla_a\widehat Q(a_0+\Delta a)
=g+H\Delta a+O(\|\Delta a\|^2).
\end{equation}
Thus
\begin{equation}
(I-\beta H)\Delta a_{\rm prox}
=\beta g+O(\beta\|\Delta a_{\rm prox}\|^2).
\end{equation}
For $\beta\|H\|<1$ and sufficiently small $\beta$,
\begin{equation}
\Delta a_{\rm prox}=\beta g+\beta^2Hg+O(\beta^3),
\end{equation}
whereas $\Delta a_{\rm lin}=\beta g$. Both are first-order schemes for fixed
positive $C$, and changing a fixed $C$ only rescales time. A positive
policy-dependent global $C$ preserves the raw-critic ascent direction under an
adaptive time parameterization in the infinitesimal limit. This derivation is
not applied to the large-budget regime with active projection, ReLU-region
crossings, critic drift, or inaccurate actor solves.

\subsection{Proof of Proposition~\ref{prop:local_composition}}
Write $x_j=x_0+jhf_0+O(h^2)$ for fixed $K$, which follows inductively from the
one-step consistency assumption. Smoothness then gives
\begin{equation}
f(x_j)=f_0+jhJ_0f_0+O(h^2).
\end{equation}
Summing the $K$ increments,
\begin{equation}
\begin{aligned}
x_K-x_0
={}&h\sum_{j=0}^{K-1}f(x_j)
+h^2\sum_{j=0}^{K-1}p(x_j)+O(Kh^3)\\
={}&Khf_0+h^2\frac{K(K-1)}{2}J_0f_0\\
&+Kh^2p_0+O(h^3).
\end{aligned}
\end{equation}
where the final remainder uses fixed $K$. Substituting $h=T/K$ proves
Eq.~\eqref{eq:k_comp}. The exact flow satisfies
$x(T)=x_0+Tf_0+\tfrac12T^2J_0f_0+O(T^3)$. Thus explicit Euler has leading
error $-T^2J_0f_0/(2K)$, while backward Euler has
$+T^2J_0f_0/(2K)$. Both remain first-order methods.

\section{Experimental Details}
\label{app:experiments}

\subsection{Algorithms and schedules}
The extension table reports the original CORL-based MPI evaluation of IQL,
ReBRAC, and TD3+BC with one or two equal-step refinements. Per-algorithm launch
grids, configurations, and model-selection metadata are not present in the
current artifact. We therefore treat the table as retrospective evidence of
refinement capacity, not as a selection-free benchmark comparison.

For the fixed-budget TD3+BC study, the total normalized budget is split as
$h=T/K$. MPI-Prox uses Eq.~\eqref{eq:actorloss} with a moving reference;
MPI-Lin constructs Eq.~\eqref{eq:linearized_target} and uses MSE regression.
The $K$ actor networks share an architecture and learning rate but are
independently initialized, persistent networks with separate Adam states.
At every second critic update, the implementation performs one optimizer step
per actor in chain order on the same minibatch. The base actor is updated first;
then its actor target and the critic target receive a Polyak update of $0.005$
before later actors use stop-gradient references from the updated online chain.
The scale $C_k$ is recomputed from each hop's reference actions. All methods
therefore match data, critic architecture, and $10^6$ critic updates, but a
$K$-hop run performs $K$ actor optimizer steps and is not compute matched.
We use batch size $256$, learning rate $3\times10^{-4}$, discount $0.99$, and
ten evaluation episodes every $5000$ updates. The 14-value grid spans $0.05$ to
$20$; the complete sweep uses two paired seeds and reports both cell-level and raw-seed failure counts.

\subsection{Diagnostic quantities}
On a fixed current-state validation batch with paired dataset actions, we
compute the empirical action-averaged proxies
\[
\begin{aligned}
\widehat D_{\rm critic}
&=\mathbb E_{(s,a)\sim\mathcal D}\frac{1}{d}\|\mu_1(s)-a\|_2^2,\\
\widehat D_{\rm final}
&=\mathbb E_{(s,a)\sim\mathcal D}\frac{1}{d}\|\mu_K(s)-a\|_2^2.
\end{aligned}
\]
These quantities measure the online actor chain relative to its sample anchor.
They are proxies for, rather than direct measurements of, the next-state
target-policy displacement in Proposition~\ref{prop:exposure}: the practical
TD target uses a Polyak target actor and policy-smoothing noise.
By Eq.~\eqref{eq:bc_decomposition}, the population expectation of each proxy
equals squared displacement from the conditional-mean behavior map plus the
same additive conditional-action variance. Population differences therefore
cancel this common floor, but ratios do not: a ratio of sample-anchored proxies
is not a ratio of mean-policy displacements. Finite validation batches add
sampling error to both interpretations.
We additionally evaluate the deterministic Polyak target actor and 64 draws
of clipped target-smoothing noise on 4096 next-state transitions per
checkpoint. All target-exposure quantities remain squared displacement per
action coordinate relative to the paired next dataset action. Thus absolute
levels include conditional action variance, whereas paired differences use a
common sampled anchor.
For MPI-Lin, projection attenuation is the RMS norm ratio between the clipped
and raw target displacements. If $\widetilde g$ is the normalized critic action
gradient and $\Delta a$ is the learned actor displacement, define
\[
h_{\rm eff}:=
\frac{\mathbb E\langle\Delta a,\widetilde g\rangle}
{\mathbb E\|\widetilde g\|_2^2}.
\]
We report $h_{\rm eff}/(dT/K)$ as the effective fraction of the nominal
linear-target coefficient. The matched audit evaluates $512$ fixed states per environment--seed run. We also record action saturation, TD-error quantiles, critic magnitude, critic loss, and gains under both the jointly trained critic and a controlled common critic: the frozen final TD3+BC $T=1$ critic matched by environment and seed. A run-total gain is the sum of mean $Q$ changes across the $K$ hops on this shared batch. The common critic supports cross-critic comparison but is not treated as ground truth. A run is descriptively labeled collapsed only from its external normalized return, using the threshold $20$.

For the exploratory simulator-reference audit, we clone MuJoCo integration
state and environment RNG from evaluation-policy trajectories, record the
episode clock, and reset the TimeLimit for a continuing-task critic-calibration
estimand. At each state we compare $a_1=\mu_1(s)$ and
$a_K=\mu_K(s)$ under continuations $\rho_1$, $\mu_1$, and $\mu_K$; the
primary calibration target here is the discounted return under the effective
backup continuation $\rho_1$. The CPU screening covers three environments,
$T\in\{4,7,10,14,20\}$, $K\in\{2,3\}$, two seeds, and 12 states per
checkpoint. It uses a 1000-step horizon, $\gamma=.99$, four common-random-number
target-smoothing rollouts, and a one-return-unit ranking dead zone. It is a
post-hoc screening study, not a causal routing intervention.

The route-only intervention trains two actor-detached shadow critics from identical initial parameters on the same minibatches and target-smoothing draws while the primary first-hop critic continues to drive the shared actor trajectory. One shadow uses the first-hop target actor and the other the final-hop target actor. The simulator audit evaluates 12 cloned states at four checkpoints in each of eight runs; inference aggregates checkpoint-level state medians to the training-run level.

The compute-matched study uses the same eight final cells. Sequential $K=3$ performs actor updates anchored at $\mu_0,\mu_1,$ and $\mu_2$; the fixed-reference control anchors both later actors at $\mu_1$; the direct-final control applies two optimizer calls to one $2T/3$ objective anchored at $\mu_1$. Every condition performs three actor optimizer calls per delayed update. The target-lag control instead keeps $K=1$ and changes only the actor-target Polyak rate from $.005$ to $.005/3$, leaving the critic-target rate unchanged.

The frozen-critic small-step audit uses the final TD3+BC $T=1$ critic for all nine environments and two seeds. At 512 interior dataset actions per run it compares the projected explicit step with a projected fixed-point solution of the implicit action equation for $h\in\{.001,.003,.01,.03,.1\}$. This is an action-level local-consistency test, not a return or coupled-training experiment.

The current-state proxy and deterministic target exposure have Pearson
correlation $.9999$ over all 270 cells; their median ratio is $.997$ and the
largest cellwise relative difference is $5.7\%$. The corresponding
correlation after target smoothing is $.9993$. The close alignment is an
empirical property of these checkpoints, where Polyak lag is small, rather
than a mathematical equivalence. Among the 84 Prox-3/Prox-2 runs in which
Prox-3 lowers deterministic target exposure, Prox-3 is nevertheless worse by
more than one score point in 17. This rejects the stronger interpretation
that minimizing exposure alone determines performance.

In the simulator-reference screening, 55 stable cells contribute 660 states
and five collapsed cells contribute 60. At $a_K$, the median signed online
critic error relative to $Q_{\rm MC}^{\rho_1}$ is $-38.7$ in stable cells and
$+1.70\times10^{12}$ in collapsed cells. This independently corroborates
the overestimation signature in Figure~\ref{fig:failure}. It does not yet
provide a reliable action-ranking rate: only 69 of 720 states have both
action differences outside the dead zone, with 29 sign disagreements, and
within-checkpoint states are correlated. Submission-level use requires
checkpoint-level uncertainty and a larger pre-specified state sample.

\section{Reproducibility Checklist}
\label{app:checklist}

\begin{enumerate}
\item \textbf{Models and algorithms.} The mathematical setting, normalized actor objective, re-centered update, scale-matched control, and critic coupling are specified in Sections~\ref{sec:method}--\ref{sec:experiments}. Base configurations follow the cited CORL implementations. Exact launchers and configurations are included for the TD3+BC fixed-budget study; original extension launch grids, configurations, and selection metadata are not archived.
\item \textbf{Theoretical claims.} Assumptions and proofs for the operator identification, finite-step margin, critic-error transfer, bootstrap-exposure bound, and local limit are provided in the main text and Appendix~\ref{app:proofs}. No true-return monotonicity is claimed.
\item \textbf{Empirical results.} The number of tasks, budgets, seeds, collapse definition, threshold sensitivity, and aggregation rules are reported. Retrospective extension results use five seeds; the complete proximal and linearized grids use two paired seeds and report raw-seed as well as seed-mean failure counts. The targeted Prox-4 comparison uses four transferred per-seed scores from a separate machine and is excluded from the complete-grid aggregate. Route, compute, lag, and frozen-critic studies report their targeted inference units and limitations.
\item \textbf{Code and assets.} Experiments use public D4RL datasets and CORL baseline implementations. The accompanying anonymized package covers the reported TD3+BC fixed-budget sweeps, controls, and diagnostic scripts. The external Prox-4 score matrices are available, but their raw evaluation files, configurations, and checkpoints have not been transferred; complete reproduction and selection auditing of the original extension analysis is also unavailable.
\item \textbf{Human subjects and sensitive data.} Not applicable.
\end{enumerate}
